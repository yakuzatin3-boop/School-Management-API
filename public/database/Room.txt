CREATE TABLE IF NOT EXISTS rooms (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    building_id UUID NOT NULL,
    floor_id UUID NOT NULL,
    room_code VARCHAR(30) NOT NULL,
    room_name VARCHAR(100),
    capacity INTEGER NOT NULL DEFAULT 30
        CHECK (capacity > 0),
    room_type VARCHAR(30) NOT NULL DEFAULT 'classroom'
        CHECK (room_type IN ('classroom', 'lab', 'meeting_room')),
    status VARCHAR(20) NOT NULL DEFAULT 'active'
        CHECK (status IN ('active', 'inactive')),

    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    FOREIGN KEY (floor_id, building_id)
        REFERENCES floors(id, building_id)
        ON DELETE RESTRICT,

    UNIQUE (building_id, room_code)
);
CREATE INDEX IF NOT EXISTS idx_floors_building_id
ON floors(building_id);

CREATE INDEX IF NOT EXISTS idx_rooms_floor_id
ON rooms(floor_id);
