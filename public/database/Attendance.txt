
CREATE TABLE IF NOT EXISTS attendance (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    enrollment_id UUID NOT NULL
        REFERENCES enrollments(id) ON DELETE RESTRICT,

    schedule_id UUID
        REFERENCES class_schedules(id) ON DELETE RESTRICT,

    attendance_date DATE NOT NULL,

    status VARCHAR(20) NOT NULL
        CHECK (status IN ('present', 'absent', 'late', 'excused')),

    marked_by UUID NOT NULL
        REFERENCES teachers(id) ON DELETE RESTRICT,

    remarks TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    UNIQUE (enrollment_id, schedule_id, attendance_date)
);

CREATE INDEX IF NOT EXISTS idx_attendance_date
ON attendance(attendance_date);

CREATE INDEX IF NOT EXISTS idx_attendance_enrollment
ON attendance(enrollment_id);
